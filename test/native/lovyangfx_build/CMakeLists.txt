cmake_minimum_required(VERSION 3.16)
project(atomlgfx_source_check LANGUAGES CXX)

get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(LGFX_SRC_ROOT "${ROOT}/third_party/LovyanGFX/src")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probe.cpp" "int atomlgfx_source_probe;\n")
add_library(probe STATIC "${CMAKE_CURRENT_BINARY_DIR}/probe.cpp")
set(COMPONENT_LIB probe)
file(SHA256 "${LGFX_SRC_ROOT}/lgfx/v1/platforms/esp32/common.inl" before)
include("${ROOT}/native/cmake/lovyangfx.cmake")
file(SHA256 "${LGFX_SRC_ROOT}/lgfx/v1/platforms/esp32/common.inl" after)
if(NOT before STREQUAL after)
    message(FATAL_ERROR "submodule の I2C ソースが変更されています")
endif()

set(unique_sources ${LGFX_SRCS})
list(REMOVE_DUPLICATES unique_sources)
list(LENGTH LGFX_SRCS source_count)
list(LENGTH unique_sources unique_count)
if(NOT source_count EQUAL unique_count)
    message(FATAL_ERROR "LovyanGFX の翻訳単位が重複しています")
endif()
foreach(source IN LISTS LGFX_SRCS)
    if(NOT EXISTS "${source}")
        message(FATAL_ERROR "LovyanGFX の翻訳単位が見つかりません: ${source}")
    endif()
endforeach()

set(platform_hub "${LGFX_SRC_ROOT}/lgfx/v1/lgfx_v1_platforms.cpp")
if(LGFX_PORT_FORCE_LEGACY_I2C)
    if(platform_hub IN_LIST LGFX_SRCS)
        message(FATAL_ERROR "旧 I2C overlay と元の hub を重複して構築しています")
    endif()
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/generated/lovyangfx/esp32_common.inl" overlay)
    if(overlay MATCHES "__has_include\\(<driver/i2c_master.h>\\)")
        message(FATAL_ERROR "旧 I2C 切り替えが適用されていません")
    endif()
elseif(NOT platform_hub IN_LIST LGFX_SRCS)
    message(FATAL_ERROR "元の platform hub が構築されません")
endif()

set(touch_hub "${LGFX_SRC_ROOT}/lgfx/v1/lgfx_v1_touch.cpp")
if(LGFX_PORT_ENABLE_TOUCH AND NOT touch_hub IN_LIST LGFX_SRCS)
    message(FATAL_ERROR "touch hub が構築されません")
elseif(NOT LGFX_PORT_ENABLE_TOUCH AND touch_hub IN_LIST LGFX_SRCS)
    message(FATAL_ERROR "無効な touch hub が構築されます")
endif()

get_target_property(definitions probe COMPILE_DEFINITIONS)
if(LGFX_PORT_DISABLE_SPI_SCT AND NOT "LGFX_SPI_SCT_DISABLE" IN_LIST definitions)
    message(FATAL_ERROR "SPI SCT の無効化が反映されません")
endif()
